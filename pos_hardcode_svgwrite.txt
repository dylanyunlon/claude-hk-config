#!/usr/bin/env python3
"""
SVG图表生成器 - 生成学术论文中的CL (Contrastive Learning) 对比图
包含功能性对比学习和优化对比学习的可视化
"""

import xml.etree.ElementTree as ET
from typing import List, Tuple, Dict, Optional

class SVGGenerator:
    """SVG图表生成器"""
    
    def __init__(self, width: int = 1200, height: int = 400):
        self.width = width
        self.height = height
        self.svg_ns = "http://www.w3.org/2000/svg"
        
        # 学术色彩调色板 - 基于seaborn的专业配色
        self.colors = {
            'gray_light': '#cccccc',
            'gray_medium': '#666666', 
            'gray_dark': '#333333',
            'node_gray': '#cccccc',
            'node_dark': '#666666',
            'text_black': '#000000',
            'stroke_black': '#000000',
            'matrix_light': '#f5f5dc',  # 浅米色
            'functionality_blue': '#4a4a9f',  # 功能性蓝色
            'functionality_purple': '#7b68aa',  # 功能性紫色  
            'functionality_gold': '#d4af82',   # 功能性金色
            'optimization_dark_blue': '#1e4d72',  # 优化深蓝
            'optimization_blue': '#4682b4',       # 优化蓝色
            'optimization_light_blue': '#b3d9ff', # 优化浅蓝
            'optimization_medium_blue': '#7bb3d9' # 优化中蓝
        }
        
    def create_svg_root(self) -> ET.Element:
        """创建SVG根元素"""
        svg = ET.Element('svg')
        svg.set('xmlns', self.svg_ns)
        svg.set('width', str(self.width))
        svg.set('height', str(self.height))
        return svg
        
    def add_defs(self, svg: ET.Element) -> ET.Element:
        """添加定义部分：样式、箭头标记和渐变"""
        defs = ET.SubElement(svg, 'defs')
        
        # 样式定义
        style = ET.SubElement(defs, 'style')
        style.text = f"""
        .node-gray {{ fill: {self.colors['node_gray']}; stroke: {self.colors['gray_medium']}; stroke-width: 2; }}
        .node-dark {{ fill: {self.colors['node_dark']}; stroke: {self.colors['gray_dark']}; stroke-width: 2; }}
        .edge {{ stroke: {self.colors['gray_dark']}; stroke-width: 2; fill: none; }}
        .connection-line {{ stroke: {self.colors['gray_dark']}; stroke-width: 1.5; fill: none; }}
        .dashed-circle {{ fill: none; stroke: {self.colors['gray_medium']}; stroke-width: 2; stroke-dasharray: 5,5; }}
        .text-label {{ font-family: Arial, sans-serif; font-size: 14px; fill: {self.colors['text_black']}; font-weight: bold; }}
        .text-normal {{ font-family: Arial, sans-serif; font-size: 11px; fill: {self.colors['text_black']}; }}
        .text-small {{ font-family: Arial, sans-serif; font-size: 10px; fill: {self.colors['text_black']}; }}
        .matrix-cell {{ stroke: {self.colors['stroke_black']}; stroke-width: 0.5; }}
        .box {{ fill: none; stroke: {self.colors['stroke_black']}; stroke-width: 1; }}
        """
        
        # 箭头标记
        marker = ET.SubElement(defs, 'marker')
        marker.set('id', 'arrowhead')
        marker.set('markerWidth', '10')
        marker.set('markerHeight', '7')
        marker.set('refX', '9')
        marker.set('refY', '3.5')
        marker.set('orient', 'auto')
        
        arrow_polygon = ET.SubElement(marker, 'polygon')
        arrow_polygon.set('points', '0 0, 10 3.5, 0 7')
        arrow_polygon.set('fill', self.colors['gray_dark'])
        
        # 功能性渐变
        grad_func = ET.SubElement(defs, 'linearGradient')
        grad_func.set('id', 'gradient_functionality')
        grad_func.set('x1', '0%')
        grad_func.set('y1', '0%')
        grad_func.set('x2', '0%')
        grad_func.set('y2', '100%')
        
        stop1 = ET.SubElement(grad_func, 'stop')
        stop1.set('offset', '0%')
        stop1.set('style', f'stop-color:{self.colors["functionality_blue"]};stop-opacity:1')
        
        stop2 = ET.SubElement(grad_func, 'stop')
        stop2.set('offset', '50%')
        stop2.set('style', f'stop-color:{self.colors["functionality_purple"]};stop-opacity:1')
        
        stop3 = ET.SubElement(grad_func, 'stop')
        stop3.set('offset', '100%')
        stop3.set('style', f'stop-color:{self.colors["functionality_gold"]};stop-opacity:1')
        
        # 优化渐变
        grad_opt = ET.SubElement(defs, 'linearGradient')
        grad_opt.set('id', 'gradient_optimization')
        grad_opt.set('x1', '0%')
        grad_opt.set('y1', '0%')
        grad_opt.set('x2', '0%')
        grad_opt.set('y2', '100%')
        
        stop1_opt = ET.SubElement(grad_opt, 'stop')
        stop1_opt.set('offset', '0%')
        stop1_opt.set('style', f'stop-color:{self.colors["optimization_dark_blue"]};stop-opacity:1')
        
        stop2_opt = ET.SubElement(grad_opt, 'stop')
        stop2_opt.set('offset', '50%')
        stop2_opt.set('style', f'stop-color:{self.colors["optimization_blue"]};stop-opacity:1')
        
        stop3_opt = ET.SubElement(grad_opt, 'stop')
        stop3_opt.set('offset', '100%')
        stop3_opt.set('style', f'stop-color:{self.colors["optimization_light_blue"]};stop-opacity:1')
        
        return defs

    def _add_connection_lines(self, g: ET.Element) -> None:
        """添加从func到O0位置的连接线"""
        # 计算关键坐标
        func_x = 405  # func文本的x坐标 (更新后的位置)
        compile_y = 85  # compile & disassemble的y坐标
        
        # 矩阵位置和O0位置
        matrix_start_x = 520
        matrix_start_y = 103
        
        # O0位置（每个格子中心）- 往下移动10个单位
        o0_positions = [
            (matrix_start_x + 9, matrix_start_y - 8),   # O0₁
            (matrix_start_x + 27, matrix_start_y - 8),  # O0₂  
            (matrix_start_x + 45, matrix_start_y - 8),  # O0₃
            (matrix_start_x + 63, matrix_start_y - 8)   # O0₄
        ]
        
        # func位置 - 更新为对应矩阵左边四个方格的中央
        func_positions = [
            (func_x, 112),  # func₁
            (func_x, 130),  # func₂
            (func_x, 148),  # func₃
            (func_x, 166)   # func₄
        ]
        
        # 为每个func添加连接线
        for i, (func_pos, o0_pos) in enumerate(zip(func_positions, o0_positions)):
            func_x, func_y = func_pos
            o0_x, o0_y = o0_pos
            
            # 计算左边线的x坐标（递减以避免重叠）
            left_x = func_x - 20 - (i * 8)
            
            # 计算上边线的y坐标（递减以避免重叠）
            up_y = compile_y - 5 - (i * 3)
            
            # 连接线路径：
            # 1. 从func左边水平向左
            line1 = ET.SubElement(g, 'line')
            line1.set('x1', str(func_x))
            line1.set('y1', str(func_y))
            line1.set('x2', str(left_x))
            line1.set('y2', str(func_y))
            line1.set('class', 'connection-line')
            
            # 2. 垂直向上
            line2 = ET.SubElement(g, 'line')
            line2.set('x1', str(left_x))
            line2.set('y1', str(func_y))
            line2.set('x2', str(left_x))
            line2.set('y2', str(up_y))
            line2.set('class', 'connection-line')
            
            # 3. 水平向右到O0位置
            line3 = ET.SubElement(g, 'line')
            line3.set('x1', str(left_x))
            line3.set('y1', str(up_y))
            line3.set('x2', str(o0_x))
            line3.set('y2', str(up_y))
            line3.set('class', 'connection-line')
            
            # 4. 垂直向下到O0
            line4 = ET.SubElement(g, 'line')
            line4.set('x1', str(o0_x))
            line4.set('y1', str(up_y))
            line4.set('x2', str(o0_x))
            line4.set('y2', str(o0_y))
            line4.set('class', 'connection-line')

    def _add_text(self, parent: ET.Element, x: str, y: str, text: str, 
                  css_class: str, anchor: str = None) -> None:
        """添加文本元素的辅助方法"""
        text_elem = ET.SubElement(parent, 'text')
        text_elem.set('x', x)
        text_elem.set('y', y)
        text_elem.set('class', css_class)
        if anchor:
            text_elem.set('text-anchor', anchor)
        text_elem.text = text
        
    def generate_svg(self) -> str:
        """生成完整的SVG字符串"""
        # 创建SVG根元素
        svg = self.create_svg_root()
        
        # 添加定义
        self.add_defs(svg)
        
        # 添加各个子图
        self.add_subfig_a(svg)
        self.add_subfig_b(svg)
        self.add_subfig_c(svg)
        self.add_subfig_d(svg)
        
        # 转换为字符串
        ET.register_namespace('', self.svg_ns)
        svg_string = ET.tostring(svg, encoding='unicode', method='xml')
        
        # 添加XML声明
        return '<?xml version="1.0" encoding="UTF-8"?>\n' + svg_string
        
    def save_svg(self, filename: str) -> None:
        """保存SVG到文件"""
        svg_content = self.generate_svg()
        with open(filename, 'w', encoding='utf-8') as f:
            f.write(svg_content)
        print(f"SVG已保存到: {filename}")
